function claudeClient(apiKey, fetchImpl = fetch) {
  return async ({ model, system, user, maxTokens = 1024, tool }) => {
    const body = {
      model,
      max_tokens: maxTokens,
      system,
      messages: [{ role: "user", content: user }]
    };
    if (tool) {
      body.tools = [tool];
      body.tool_choice = { type: "tool", name: tool.name };
    }
    const send = () => fetchImpl("https://api.anthropic.com/v1/messages", {
      method: "POST",
      headers: {
        "content-type": "application/json",
        "x-api-key": apiKey,
        "anthropic-version": "2023-06-01"
      },
      body: JSON.stringify(body)
    });
    let res = await send();
    if (!res.ok) {
      let detail = await res.text().catch(() => "");
      if (res.status === 400 && tool && /tool_choice/.test(detail)) {
        body.tool_choice = { type: "auto" };
        res = await send();
        if (!res.ok) detail = await res.text().catch(() => "");
      }
      if (!res.ok) throw new Error(`Claude API svarede ${res.status}: ${detail.slice(0, 300)}`);
    }
    const data = await res.json();
    const blocks = data.content ?? [];
    const text = blocks.filter((b) => b.type === "text").map((b) => String(b.text)).join("").trim();
    const toolBlock = blocks.find((b) => b.type === "tool_use");
    return { text, toolInput: toolBlock?.input ?? null };
  };
}

var ADDRESS = /[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}/gi;
function detectPlatform(mail) {
  const candidates = [mail.from, ...forwardedSenders(mail.text)];
  for (const addr of candidates) {
    for (const [re, platform] of SENDER_DOMAINS) if (re.test(addr)) return platform;
  }
  return "other";
}
function forwardedSenders(text) {
  const out = [];
  const lines = text.split(/\r?\n/).slice(0, 200);
  for (const line of lines) {
    if (/^\s*(>\s*)*(from|fra|von|de|van)\s*:/i.test(line)) out.push(...line.match(ADDRESS) ?? []);
  }
  return out;
}
function inboundToken(to, mailDomain) {
  const domain = mailDomain.toLowerCase();
  for (const addr of to.match(ADDRESS) ?? []) {
    const [local, host] = addr.toLowerCase().split("@");
    if (host !== domain) continue;
    const token = local.split("+")[0];
    if (/^[0-9a-f]{16}$/.test(token)) return token;
  }
  return null;
}
function cleanEmailText(text, max = 12e3) {
  return text.replace(/\r/g, "").replace(/[ \t]+\n/g, "\n").replace(/\n{3,}/g, "\n\n").trim().slice(0, max);
}

function forwardingConfirmation(mail) {
  const senders = [mail.from, ...forwardedSenders(mail.text)].join(" ").toLowerCase();
  if (!senders.includes("forwarding-noreply@google.com")) return null;
  const all = `${mail.subject}
${mail.text}`;
  const code = all.match(/\(#(\d{6,12})\)/)?.[1] ?? all.match(/(?:code|kode|Code)[^0-9]{0,40}(\d{6,12})/i)?.[1] ?? null;
  const url = all.match(/https:\/\/mail(?:-settings)?\.google\.com\/mail\/[^\s<>"')]+/)?.[0] ?? null;
  const account = all.match(/(?:from|fra)\s+([A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,})/i)?.[1] ?? null;
  return { provider: "gmail", code, confirmUrl: url, account };
}
async function dedupeKey(r) {
  if (r.externalId) return `${r.platform}:id:${r.externalId}`;
  const basis = [
    r.platform,
    r.propertyId ?? "",
    (r.author ?? "").trim().toLowerCase(),
    (r.publishedAt ?? "").slice(0, 10),
    (r.body ?? r.title ?? "").replace(/\s+/g, " ").trim().toLowerCase().slice(0, 160)
  ].join("|");
  const digest = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(basis));
  const hex = [...new Uint8Array(digest)].map((b) => b.toString(16).padStart(2, "0")).join("");
  return `${r.platform}:h:${hex.slice(0, 32)}`;
}

function normalizeExtraction(input, platform, properties) {
  const s = (v) => typeof v === "string" && v.trim() ? v.trim() : null;
  const n = (v) => typeof v === "number" && Number.isFinite(v) ? v : null;
  const i = input ?? {};
  let scale = n(i.scale);
  if (scale !== 5 && scale !== 6 && scale !== 10) scale = null;
  scale ??= PLATFORM_SCALE[platform] ?? null;
  let rating = n(i.rating);
  if (rating !== null && (rating < 0 || scale === null || rating > scale)) rating = null;
  let propertyId = s(i.property_id);
  if (propertyId && !properties.some((p) => p.id === propertyId)) propertyId = null;
  if (!propertyId && properties.length === 1) propertyId = properties[0].id;
  const published = s(i.published_at);
  return {
    isReview: i.is_review === true,
    platform,
    propertyId,
    propertyHint: s(i.property_name)?.slice(0, 120) ?? null,
    author: s(i.author)?.slice(0, 120) ?? null,
    rating,
    scale,
    title: s(i.title)?.slice(0, 300) ?? null,
    body: guestText(s(i.body))?.slice(0, 8e3) ?? null,
    language: s(i.language)?.slice(0, 8) ?? null,
    publishedAt: published && /^\d{4}-\d{2}-\d{2}/.test(published) ? published.slice(0, 10) : null,
    externalId: s(i.external_id)?.slice(0, 200) ?? null,
    sourceUrl: (() => {
      const u = s(i.source_url);
      return u && /^https:\/\//.test(u) ? u : null;
    })(),
    hasCriticism: i.has_criticism === true,
    isExcerpt: i.is_excerpt === true
  };
}
var PLATFORM_PHRASES = [
  /det ser ud til,? at de havde et (glimrende|godt|dejligt) ophold/i,
  /tak,? fordi du giver .{0,60}(en god|gode) oplevelse/i,
  /giv dine gæster et godt indtryk/i,
  /it looks like they (had|enjoyed) (a|an) (great|excellent|good) stay/i,
  /thanks? for (giving|providing) .{0,60}(a )?great (experience|stay)/i,
  /^you have a new (guest )?review/i
];
function guestText(body) {
  if (!body) return null;
  let t = body;
  for (const re of PLATFORM_PHRASES) {
    t = t.split(/(?<=[.!?])\s+/).filter((sentence) => !re.test(sentence)).join(" ");
  }
  t = t.trim();
  return t.length >= 2 ? t : null;
}
async function extractReview(claude, model, mail, properties) {
  const platform = detectPlatform(mail);
  const { system, user } = extractionPrompt(mail, platform, properties);
  const { toolInput } = await claude({ model, system, user, tool: EXTRACT_TOOL, maxTokens: 2e3 });
  return normalizeExtraction(toolInput, platform, properties);
}

function cleanReply(text) {
  let t = text.trim();
  t = t.replace(/^(svar|reply|antwort)\s*:\s*/i, "");
  if (/^["“].*["”]$/s.test(t)) t = t.slice(1, -1).trim();
  return t.replace(/\n{3,}/g, "\n\n").slice(0, 4e3);
}
async function draftReply(claude, model, ctx) {

  if (!reply) throw new Error("AI'en returnerede et tomt svar");
  return reply;
}

async function ingestEmail(mail, deps) {
  const token = inboundToken(mail.to, deps.mailDomain);
  const org = token ? await deps.findOrgByToken(token) : null;
  if (!org) {
    await deps.logInbound({ to_address: mail.to.slice(0, 300), from_address: mail.from.slice(0, 300), status: "unknown_recipient" });
    return { status: "unknown_recipient" };
  }
  const inboundId = await deps.logInbound({
    org_id: org.id,
    to_address: mail.to.slice(0, 300),
    from_address: mail.from.slice(0, 300),
    subject: mail.subject.slice(0, 500),
    text_body: mail.text.slice(0, 5e4),
    status: "received"
  });
  const confirmation = forwardingConfirmation(mail);
  if (confirmation) {
    await deps.saveConfirmation(org.id, confirmation);
    await deps.updateInbound(inboundId, { status: "ignored", error: "Bekr\xE6ftelse af videresendelse fra Gmail", text_body: null });
    return { status: "confirmation", inboundId };
  }
  if (!looksLikeReview(mail)) {
    await deps.updateInbound(inboundId, { status: "ignored", error: "Ikke en anmeldelse", text_body: null });
    return { status: "ignored", reason: "Mailen indeholder ikke en anmeldelse" };
  }
  if (!org.active) {
    await deps.updateInbound(inboundId, { status: "ignored", error: "Abonnementet er udl\xF8bet" });
    return { status: "ignored", reason: "Abonnementet er udl\xF8bet" };
  }
  try {
    const properties = await deps.listProperties(org.id);
    const review = await extractReview(deps.claude, deps.models.extract, mail, properties);
    if (!review.isReview) {
      await deps.updateInbound(inboundId, { status: "ignored", platform: review.platform, text_body: null });
      return { status: "ignored", reason: "Mailen indeholder ikke en anmeldelse" };
    }
    const saved = await deps.insertReview({
      org_id: org.id,
      property_id: review.propertyId,
      property_hint: review.propertyId ? null : review.propertyHint,
      platform: review.platform,
      external_id: review.externalId,
      dedupe_key: await dedupeKey(review),
      author: review.author,
      rating: review.rating,
      scale: review.scale,
      title: review.title,
      body: review.body,
      language: review.language,
      published_at: review.publishedAt,
      published_precision: review.publishedAt ? "day" : "unknown",
      source: "email",
      source_url: review.sourceUrl,
      needs_approval: review.hasCriticism,
      raw: { inbound_id: inboundId, excerpt: review.isExcerpt }
    });
    if (!saved) {
      await deps.updateInbound(inboundId, { status: "ignored", platform: review.platform, error: "Anmeldelsen findes allerede" });
      return { status: "duplicate", inboundId };
    }
    await deps.updateInbound(inboundId, { status: "parsed", platform: review.platform, review_id: saved.id });
    let drafted = false;
    if (["unanswered", "awaiting_approval"].includes(saved.reply_status) && (review.body || review.title)) {
      const property = properties.find((p) => p.id === review.propertyId) ?? null;
      try {
        const body = await draftReply(deps.claude, deps.models.reply, {
          orgName: org.name,
          propertyName: property?.name ?? null,
          signoff: property?.signoff ?? org.signoff,
          toneExamples: await deps.toneExamples(org.id, review.propertyId),
          notes: [],
          review: {
            platform: review.platform,
            author: review.author,
            rating: review.rating,
            scale: review.scale,
            title: review.title,
            body: review.body,
            language: review.language,
            needsApproval: review.hasCriticism
          }
        });
        await deps.insertDraft({ org_id: org.id, review_id: saved.id, body, origin: "ai", model: deps.models.reply });
        drafted = true;
      } catch (_e) {
      }
    }
    return { status: "parsed", inboundId, reviewId: saved.id, drafted };
  } catch (e) {
    const error = e instanceof Error ? e.message : String(e);
    await deps.updateInbound(inboundId, { status: "failed", error: error.slice(0, 500) });
    return { status: "failed", inboundId, error };
  }
}

function safeEqual(a, b) {
  const x = new TextEncoder().encode(a), y = new TextEncoder().encode(b);
  if (x.length !== y.length) return false;
  let diff = 0;
  for (let i = 0; i < x.length; i++) diff |= x[i] ^ y[i];
  return diff === 0;
}

Deno.serve(async (req) => {
  if (req.method !== "POST") return json({ error: "Kun POST" }, 405);
  const secret = Deno.env.get("INGEST_SECRET") ?? "";
  if (!secret || !safeEqual(req.headers.get("x-ingest-secret") ?? "", secret)) {
    return json({ error: "Ikke tilladt" }, 401);
  }
  let mail;
  try {
    mail = await req.json();
  } catch {
    return json({ error: "Ugyldig JSON" }, 400);
  }
  if (!mail.to || !mail.text) return json({ error: "Mangler to eller text" }, 400);
  const db = serviceClient();
  const result = await ingestEmail(
    { to: mail.to, from: mail.from ?? "", subject: mail.subject ?? "", text: mail.text },
    {
      mailDomain: Deno.env.get("INBOUND_MAIL_DOMAIN"),
      claude: claudeClient(Deno.env.get("ANTHROPIC_API_KEY")),
      models: MODELS,
      async findOrgByToken(token) {
        const { data } = await db.from("organizations").select("id, name, signoff, subscription_status, trial_ends_at").eq("inbound_token", token).maybeSingle();
        if (!data) return null;
        const active = ["active", "pilot"].includes(data.subscription_status) || data.subscription_status === "trialing" && new Date(data.trial_ends_at) > /* @__PURE__ */ new Date();
        return { id: data.id, name: data.name, signoff: data.signoff, active };
      },
      async listProperties(orgId) {
        const { data, error } = await db.from("properties").select("id, name, aliases, signoff").eq("org_id", orgId).is("archived_at", null);
        if (error) throw error;
        return data ?? [];
      },
      async logInbound(row) {
        const { data, error } = await db.from("inbound_emails").insert(row).select("id").single();
        if (error) throw error;
        return data.id;
      },
      async updateInbound(id, patch) {
        const { error } = await db.from("inbound_emails").update(patch).eq("id", id);
        if (error) throw error;
      },
      async insertReview(row) {
        const { data, error } = await db.from("reviews").upsert(row, { onConflict: "org_id,dedupe_key", ignoreDuplicates: true }).select("id, reply_status");
        if (error) throw error;
        return data?.[0] ?? null;
      },
      async toneExamples(orgId, propertyId) {
        let q = db.from("tone_examples").select("body").eq("org_id", orgId).order("created_at", { ascending: false }).limit(8);
        if (propertyId) q = q.or(`property_id.eq.${propertyId},property_id.is.null`);
        const { data } = await q;
        return (data ?? []).map((r) => r.body);
      },
      async insertDraft(row) {
        const { error } = await db.from("reply_drafts").insert(row);
        if (error) throw error;
      },
      async saveConfirmation(orgId, c) {
        const { error } = await db.from("forwarding_confirmations").insert({
          org_id: orgId,
          provider: c.provider,
          code: c.code,
          confirm_url: c.confirmUrl,
          account: c.account
        });
        if (error) throw error;
      }
    }
  );
  return json(result, 200);
});
